int power(int n, int a) //n的a次方

{

    int i = 0;
    
    int num = 1;
    
    while (i < a)
    
    {
    
        num *= n;
        
        i++;
        
    }
    
    return num;
    
}

void turn(Stack* S, int way1, char* num, int way2)//主要功能函数

{

    everyone_worth(S, num);
        
    int sum = 0, way = 0, to_save;//to_save，用于存储待转换数每一位10进制的值
        
    char n;
        
    char* p = &n;
    
    while (StackEmpty(S) != 1)
        
    {
        
        Pop(S, p);
            
        if (p != NULL)
            
        {
            
            if ('0' <= *p && *p <= '9')
                
            {
                
                to_save = (int)*p - 48;
                
            }
                
            if ('A' <= *p && *p <= 'Z')
                
            {
                
                to_save = (int)*p - 55;
                
            }
                
            sum += to_save * power(way1, way);//这下转换成十进制了，sum就是了
                
            way++;
            
        }
        
    }//sum是该数十进制的大小，接下来转换为way2进制，只需要取余进栈，结束条件为0

    if(way2==10)

        printf("%s由%d进制转化为%d进制，其值为：%d", num, way1, way2,sum);
        
        return;
        
    }
